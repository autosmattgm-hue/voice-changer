  79 | function plainTextAiOutput(content) {
  80 |   return String(content || "")
  81 |     .replace(/\r\n/g, "\n")
  82 |     .split("\n")
  83 |     .map((line) => line
  84 |       .replace(/^\s{0,3}#{1,6}\s*/g, "")
  85 |       .replace(/^\s*[*]+\s+/g, "")
  86 |       .replace(/^\s*[-]\s+/g, "")
  87 |       .replace(/\*\*(.*?)\*\*/g, "$1")
  88 |       .replace(/\*(.*?)\*/g, "$1")
  89 |       .replace(/__([^_]+)__/g, "$1")
  90 |       .replace(/`([^`]+)`/g, "$1")
  91 |       .replace(/[#$*]/g, "")
  92 |       .trimEnd())
  93 |     .join("\n")
  94 |     .replace(/\n{3,}/g, "\n\n")
  95 |     .trim();
  96 | }

 108 | function compactObject(value) {
 109 |   return Object.fromEntries(Object.entries(value || {}).filter(([, item]) => {
 110 |     if (item === undefined || item === null || item === "") return false;
 111 |     if (Array.isArray(item) && item.length === 0) return false;
 112 |     if (typeof item === "object" && !Array.isArray(item) && Object.keys(item).length === 0) return false;
 113 |     return true;
 114 |   }));
 115 | }
