   1 | // Helpers to judge whether an AI-built website page contains real, *visible*
   2 | // content (not just a <style> block). A model that returns only CSS is a failed
   3 | // build — these helpers let the builders reject it, retry, or fall back to the
   4 | // content-rich template fragment so the shipped site is never empty.
   5 | 
   6 | // Remove everything that renders no text: <style>, <script> and comments.
   7 | export function stripInert(html) {
   8 |   return String(html || "")
   9 |     .replace(/<style[\s\S]*?<\/style>/gi, " ")
  10 |     .replace(/<script[\s\S]*?<\/script>/gi, " ")
  11 |     .replace(/<!--[\s\S]*?-->/g, " ");
  12 | }
  13 | 
  14 | // Visible, human-readable text with tags and entities collapsed to spaces.
  15 | export function visibleText(html) {
  16 |   return stripInert(html)
  17 |     .replace(/<[^>]+>/g, " ")
  18 |     .replace(/&[a-z]+;/gi, " ")
  19 |     .replace(/\s+/g, " ")
  20 |     .trim();
  21 | }
  22 | 
  23 | export function visibleWordCount(html) {
  24 |   const text = visibleText(html);
  25 |   return text ? text.split(" ").filter(Boolean).length : 0;
  26 | }
  27 | 
  28 | // A single page fragment counts as real when it has visible prose AND at least
  29 | // one heading. A CSS-only block has neither, so it fails and triggers the
  30 | // template fallback fragment (which always carries real copy).
  31 | export function hasRealContent(fragment, { minWords = 40 } = {}) {
  32 |   const raw = String(fragment || "");
  33 |   if (!/<h[1-6][\s>]/i.test(raw)) return false;
  34 |   return visibleWordCount(raw) >= minWords;
  35 | }
  36 | 
  37 | // Whole-document check used by WebsiteService.build(): the assembled shell always
  38 | // contributes nav + footer words, so require enough visible words that only a
  39 | // genuine multi-section page reaches the threshold. A CSS-only shell lands near
  40 | // the nav/footer word count and is rejected.
  41 | export function hasRichVisibleContent(html, { minWords = 70 } = {}) {
  42 |   if (!/<\/html>/i.test(String(html || ""))) return false;
  43 |   return visibleWordCount(html) >= minWords;
  44 | }
