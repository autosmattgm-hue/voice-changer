  83 | function publicPollinationsError(error) {
  84 |   if (error instanceof AppError) return error;
  85 |   const m = String(error?.message || "").toLowerCase();
  86 |   if (m.includes("end of life") || m.includes("end-of-life") || m.includes("decommissioned") || m.includes("retired") || m.includes("no longer supported") || m.includes("not found")) {
  87 |     return new AppError("The AI provider endpoint is no longer available. Check POLLINATIONS_API_KEY and retry.", 503, "POLLINATIONS_EOL");
  88 |   }
  89 |   if (m.includes("401") || m.includes("403") || m.includes("unauthorized") || m.includes("forbidden")) {
  90 |     return new AppError("The AI provider key is missing or invalid. Add POLLINATIONS_API_KEY to .env and restart.", 503, "POLLINATIONS_AUTH_FAILED");
  91 |   }
  92 |   if (m.includes("402") || m.includes("quota") || m.includes("insufficient") || m.includes("credit")) {
  93 |     return new AppError("The AI provider free quota is exceeded. Add a valid key or check your balance.", 503, "POLLINATIONS_QUOTA_EXCEEDED");
  94 |   }
  95 |   if (m.includes("429") || m.includes("rate limit") || m.includes("too many requests")) {
  96 |     return new AppError("The AI provider is rate-limiting this request. Pause and retry.", 503, "POLLINATIONS_RATE_LIMITED");
  97 |   }
  98 |   if (String(error?.name || "").toLowerCase() === "aborterror" || /timed out/i.test(String(error?.message || ""))) {
  99 |     return new AppError("The AI provider timed out. Please retry.", 504, "POLLINATIONS_TIMEOUT");
 100 |   }
 101 |   if (m.includes("404") || m.includes("not found") || m.includes("model not found") || m.includes("endpoint not found")) {
 102 |     return new AppError("The AI provider returned a 404. Check the configured model and retry.", 503, "POLLINATIONS_NOT_FOUND");
 103 |   }
 104 |   return new AppError("The AI provider request failed. Click Generate again to retry.", 503, "POLLINATIONS_UNAVAILABLE");
 105 | }

 107 | function fetchJson(url, options) {
 108 |   return fetch(url, options).then(async (response) => {
 109 |     const text = await response.text();
 110 |     let payload = null;
 111 |     const ct = response.headers.get("content-type") || "";
 112 |     if (ct.includes("application/json")) {
 113 |       try {
 114 |         payload = JSON.parse(text);
 115 |       } catch {
 116 |         payload = { raw: text };
 117 |       }
 118 |     } else {
 119 |       payload = { raw: text };
 120 |     }
 121 |     if (!response.ok) {
 122 |       const status = response.status;
 123 |       const message = status === 401 || status === 403 ? "Unauthorized" : text ? String(text).slice(0, 200) : "Request failed";
 124 |       const error = new Error(message);
 125 |       error.name = "POLLINATIONS_" + status;
 126 |       error.status = status;
 127 |       error.code = "POLLINATIONS_" + status;
 128 |       throw error;
 129 |     }
 130 |     return payload;
 131 |   });
