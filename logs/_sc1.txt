  26 | function stringField(input, field, options = {}) {
  27 |   const value = input[field] === undefined || input[field] === null ? "" : String(input[field]).trim();
  28 |   const errors = [];
  29 | 
  30 |   if (options.required && !value) errors.push("Required");
  31 |   if (options.min && value.length < options.min) errors.push(`Must be at least ${options.min} characters`);
  32 |   if (options.max && value.length > options.max && !options.truncate) errors.push(`Must be at most ${options.max} characters`);
  33 |   if (options.email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) errors.push("Must be a valid email");
  34 | 
  35 |   if (errors.length) throw validationError({ [field]: errors });
  36 |   const normalized = options.max && options.truncate ? value.slice(0, options.max) : value;
  37 |   return normalized || options.default || "";
  38 | }

  40 | function stringListField(input, field, options = {}) {
  41 |   const raw = input[field];
  42 |   let values = [];
  43 |   if (Array.isArray(raw)) {
  44 |     values = raw;
  45 |   } else if (typeof raw === "string") {
  46 |     // Browsers send multi-selects as comma strings; dashboard also sends arrays.
  47 |     values = raw.split(/[,;\n|]/);
  48 |   } else if (raw !== undefined && raw !== null && raw !== "") {
  49 |     values = [String(raw)];
  50 |   }
  51 |   const cleaned = [...new Set(values
  52 |     .map((value) => String(value || "").trim())
  53 |     .filter(Boolean))]
  54 |     .slice(0, options.maxItems || 20);
  55 | 
  56 |   for (const value of cleaned) {
  57 |     if (options.max && value.length > options.max) {
  58 |       throw validationError({ [field]: [`Each value must be at most ${options.max} characters`] });
  59 |     }
  60 |   }
  61 | 
  62 |   return cleaned;
  63 | }

  65 | function numberField(input, field, options = {}) {
  66 |   const value = input[field] === undefined || input[field] === "" ? options.default : Number(input[field]);
  67 |   if (!Number.isFinite(value)) {
  68 |     if (options.defaultOnInvalid) return options.default;
  69 |     throw validationError({ [field]: ["Must be a number"] });
  70 |   }
  71 |   const normalized = options.int ? Math.round(value) : value;
  72 |   if (options.min !== undefined && normalized < options.min) {
  73 |     if (options.clamp) return options.min;
  74 |     throw validationError({ [field]: [`Must be at least ${options.min}`] });
  75 |   }
  76 |   if (options.max !== undefined && normalized > options.max) {
  77 |     if (options.clamp) return options.max;
  78 |     throw validationError({ [field]: [`Must be at most ${options.max}`] });
  79 |   }
  80 |   return normalized;
  81 | }

  83 | function booleanField(input, field, options = {}) {
  84 |   const value = input[field];
  85 |   if (value === undefined || value === null || value === "") return Boolean(options.default);
  86 |   if (typeof value === "boolean") return value;
  87 |   if (typeof value === "number") return value === 1;
  88 |   const normalized = String(value).trim().toLowerCase();
  89 |   if (["true", "1", "yes", "on"].includes(normalized)) return true;
  90 |   if (["false", "0", "no", "off"].includes(normalized)) return false;
  91 |   return Boolean(options.default);
  92 | }

  96 |   const value = Number(input[field]);

  98 |     throw validationError({ [field]: [`Must be between ${min} and ${max}`] });
  99 |   }

 104 |   const value = input[field] || options.default;
 105 |   if (!allowed.includes(value)) throw validationError({ [field]: [`Must be one of: ${allowed.join(", ")}`] });
